     1→"use client";
     2→
     3→import { useState, useEffect, useMemo, useCallback } from "react";
     4→import { useRouter } from "next/navigation";
     5→import { motion, AnimatePresence } from "framer-motion";
     6→import { Button } from "@/components/ui/button";
     7→import { Card } from "@/components/ui/card";
     8→import { Badge } from "@/components/ui/badge";
     9→import { useGameStore } from "@/lib/game-store";
    10→import { Level, Word } from "@/lib/game-data";
    11→import { CrosswordGrid } from "./crossword-grid";
    12→import { fetchLevelById, saveProgress } from "@/lib/game-data-api";
    13→import { useToast } from "@/hooks/use-toast";
    14→import { playSound, getMuted, setMuted } from "@/lib/sound-effects";
    15→import { cn } from "@/lib/utils";
    16→
    17→// الأبجدية العربية الكاملة مرتبة
    18→const ARABIC_KEYBOARD = [
    19→  ["ض", "ص", "ث", "ق", "ف", "غ", "ع", "ه", "خ", "ح", "ج", "د"],
    20→  ["ش", "س", "ي", "ب", "ل", "ا", "ت", "ن", "م", "ك", "ط", "ئ"],
    21→  ["ء", "ؤ", "ر", "لا", "ى", "ة", "و", "ز", "ظ", "ذ"],
    22→];
    23→
    24→interface GameScreenProps {
    25→  levelId: number;
    26→  onExit?: () => void;
    27→}
    28→
    29→export function GameScreen({ levelId, onExit }: GameScreenProps) {
    30→  const router = useRouter();
    31→  const { toast } = useToast();
    32→  const [level, setLevel] = useState<Level | null>(null);
    33→  const [loading, setLoading] = useState(true);
    34→
    35→  const { completeLevel, addCoins, spendCoins, useHint: recordHintUsed, resetLevelStats, coins, hintsUsed: globalHintsUsed, getLevelStars, userId } =
    36→    useGameStore();
    37→
    38→  const [foundWords, setFoundWords] = useState<string[]>([]);
    39→  const [bonusWords, setBonusWords] = useState<string[]>([]);
    40→  const [showSuccess, setShowSuccess] = useState(false);
    41→  const [startTime, setStartTime] = useState<number>(Date.now());
    42→  const [isMuted, setIsMuted] = useState(false);
    43→  const [currentWordIndex, setCurrentWordIndex] = useState(0);
    44→  const [selectedLetters, setSelectedLetters] = useState<{ letter: string; used: boolean }[]>([]);
    45→  const [answerOrder, setAnswerOrder] = useState<number[]>([]); // أرقام index في selectedLetters بترتيب النقر
    46→  const [revealedLetters, setRevealedLetters] = useState<Set<number>>(new Set());
    47→  const [wrongShake, setWrongShake] = useState(false);
    48→  const [levelHintsUsed, setLevelHintsUsed] = useState(0); // تلميحات المرحلة الحالية فقط
    49→
    50→  useEffect(() => {
    51→    setIsMuted(getMuted());
    52→  }, []);
    53→
    54→  useEffect(() => {
    55→    let cancelled = false;
    56→    setLoading(true);
    57→    setFoundWords([]);
    58→    setBonusWords([]);
    59→    setShowSuccess(false);
    60→    setStartTime(Date.now());
    61→    setCurrentWordIndex(0);
    62→    setSelectedLetters([]);
    63→    setAnswerOrder([]);
    64→    setRevealedLetters(new Set());
    65→    setLevelHintsUsed(0); // إعادة تعيين تلميحات المرحلة
    66→
    67→    fetchLevelById(levelId).then((data) => {
    68→      if (!cancelled) {
    69→        setLevel(data);
    70→        setLoading(false);
    71→        if (data) {
    72→          setTimeout(() => playSound("levelStart"), 200);
    73→          // تهيئة الحروف المتاحة مع لوحة المفاتيح الكاملة
    74→          if (data.availableLetters && data.availableLetters.length > 0) {
    75→            const letters = data.availableLetters.map(l => ({ letter: l, used: false }));
    76→            setSelectedLetters(letters);
    77→          }
    78→        }
    79→      }
    80→    });
    81→    return () => { cancelled = true; };
    82→  }, [levelId]);
    83→
    84→  const currentWord = level?.words[currentWordIndex];
    85→  const currentWordText = currentWord?.text || "";
    86→
    87→  // الحروف المعروضة في خانات الإجابة (مكشوفة + مختارة من المستخدم بترتيب النقر)
    88→  const answerLetters = useMemo(() => {
    89→    if (!currentWord) return [];
    90→    const letters: (string | null)[] = Array(currentWordText.length).fill(null);
    91→    // ملء الحروف المكتشفة بالتلميح أولاً
    92→    revealedLetters.forEach((idx) => {
    93→      if (idx < currentWordText.length) {
    94→        letters[idx] = currentWordText[idx];
    95→      }
    96→    });
    97→    // ملء الحروف المختارة من المستخدم بترتيب النقر
    98→    let letterIdx = 0;
    99→    for (const selIdx of answerOrder) {
   100→      // إيجاد أول خانة فارغة
   101→      while (letterIdx < currentWordText.length && letters[letterIdx] !== null) {
   102→        letterIdx++;
   103→      }
   104→      if (letterIdx < currentWordText.length && selectedLetters[selIdx]) {
   105→        letters[letterIdx] = selectedLetters[selIdx].letter;
   106→        letterIdx++;
   107→      }
   108→    }
   109→    return letters;
   110→  }, [currentWord, currentWordText, revealedLetters, selectedLetters, answerOrder]);
   111→
   112→  const toggleMute = () => {
   113→    const newMuted = !isMuted;
   114→    setIsMuted(newMuted);
   115→    setMuted(newMuted);
   116→  };
   117→
   118→  const handleLetterClick = useCallback((letter: string) => {
   119→    // البحث عن أول حرف غير مستخدم مطابق
   120→    const idx = selectedLetters.findIndex(l => l.letter === letter && !l.used);
   121→    if (idx === -1) return;
   122→
   123→    const newLetters = [...selectedLetters];
   124→    newLetters[idx] = { ...newLetters[idx], used: true };
   125→    const newOrder = [...answerOrder, idx];
   126→    setSelectedLetters(newLetters);
   127→    setAnswerOrder(newOrder);
   128→    playSound("selectLetter");
   129→
   130→    // حساب عدد الخانات المملوءة بعد هذا الاختيار
   131→    const filledCount = revealedLetters.size + newOrder.length;
   132→    if (filledCount >= currentWordText.length) {
   133→      const word = buildCurrentWord(newLetters, newOrder);
   134→      setTimeout(() => checkWord(word, newLetters, newOrder), 200);
   135→    }
   136→  }, [selectedLetters, answerOrder, revealedLetters, currentWordText]);
   137→
   138→  const buildCurrentWord = (letters: { letter: string; used: boolean }[], order: number[]): string => {
   139→    if (!currentWord) return "";
   140→    const result: (string | null)[] = Array(currentWordText.length).fill(null);
   141→    // ملء الحروف المكتشفة بالتلميح
   142→    revealedLetters.forEach((idx) => {
   143→      if (idx < currentWordText.length) {
   144→        result[idx] = currentWordText[idx];
   145→      }
   146→    });
   147→    // ملء الحروف المختارة من المستخدم بترتيب النقر
   148→    let letterIdx = 0;
   149→    for (const selIdx of order) {
   150→      while (letterIdx < currentWordText.length && result[letterIdx] !== null) {
   151→        letterIdx++;
   152→      }
   153→      if (letterIdx < currentWordText.length && letters[selIdx]) {
   154→        result[letterIdx] = letters[selIdx].letter;
   155→        letterIdx++;
   156→      }
   157→    }
   158→    return result.join("");
   159→  };
   160→
   161→  const checkWord = (word: string, letters: { letter: string; used: boolean }[], order: number[]) => {
   162→    if (!level || !currentWord) return;
   163→
   164→    const isCorrect = word === currentWordText;
   165→    const isBonus = level.bonusWords.includes(word) && !bonusWords.includes(word);
   166→
   167→    if (isCorrect) {
   168→      const newFound = [...foundWords, word];
   169→      setFoundWords(newFound);
   170→      playSound("correctWord");
   171→      toast({ title: "🎉 إجابة صحيحة!", description: currentWord.clue });
   172→
   173→      setTimeout(() => {
   174→        if (currentWordIndex + 1 < level.words.length) {
   175→          setCurrentWordIndex(currentWordIndex + 1);
   176→          setRevealedLetters(new Set());
   177→          // إعادة تعيين الحروف المستخدمة
   178→          setSelectedLetters(prev => prev.map(l => ({ ...l, used: false }))); setAnswerOrder([]);
   179→        } else {
   180→          playSound("levelComplete");
   181→          setShowSuccess(true);
   182→          completeLevel(level.id, level.reward);
   183→          addCoins(bonusWords.length * 20);
   184→          const timeSpent = Math.floor((Date.now() - startTime) / 1000);
   185→          if (userId) {
   186→            saveProgress({
   187→              userId, levelId: level.id, isCompleted: true, stars: 1,
   188→              score: level.reward + bonusWords.length * 20,
   189→              hintsUsed: levelHintsUsed, timeSpent,
   190→              coinsEarned: level.reward + bonusWords.length * 20,
   191→            }).catch((e) => console.error("Cloud sync error:", e));
   192→          }
   193→        }
   194→      }, 800);
   195→    } else if (isBonus) {
   196→      setBonusWords([...bonusWords, word]);
   197→      addCoins(20);
   198→      playSound("bonusWord");
   199→      toast({ title: "✨ كلمة إضافية!", description: `+20 عملة (${word})` });
   200→      setSelectedLetters(prev => prev.map(l => ({ ...l, used: false }))); setAnswerOrder([]);
   201→    } else {
   202→      playSound("wrongWord");
   203→      setWrongShake(true);
   204→      setTimeout(() => setWrongShake(false), 500);
   205→      toast({ title: "❌ إجابة خاطئة", description: "حاول مرة أخرى", variant: "destructive" });
   206→      setTimeout(() => {
   207→        setSelectedLetters(prev => prev.map(l => ({ ...l, used: false }))); setAnswerOrder([]);
   208→      }, 500);
   209→    }
   210→  };
   211→
   212→  const handleRemoveLast = () => {
   213→    if (answerOrder.length === 0) return;
   214→    // إزالة آخر حرف من ترتيب النقر
   215→    const lastIdx = answerOrder[answerOrder.length - 1];
   216→    const newOrder = answerOrder.slice(0, -1);
   217→    const newLetters = [...selectedLetters];
   218→    newLetters[lastIdx] = { ...newLetters[lastIdx], used: false };
   219→    setSelectedLetters(newLetters);
   220→    setAnswerOrder(newOrder);
   221→    playSound("removeLetter");
   222→  };
   223→
   224→  const handleClear = () => {
   225→    setSelectedLetters(prev => prev.map(l => ({ ...l, used: false }))); setAnswerOrder([]);
   226→    playSound("removeLetter");
   227→  };
   228→
   229→  const handleRevealLetter = () => {
   230→    if (!level || !currentWord) return;
   231→    if (coins < level.hintCost) {
   232→      toast({ title: "عملات غير كافية", description: `تحتاج ${level.hintCost} عملة`, variant: "destructive" });
   233→      return;
   234→    }
   235→    const emptyIdx = answerLetters.findIndex(l => l === null);
   236→    if (emptyIdx === -1) {
   237→      toast({ title: "اكتملت الكلمة", description: "اضغط تأكيد للتحقق" });
   238→      return;
   239→    }
   240→    if (spendCoins(level.hintCost)) {
   241→      recordHintUsed();
   242→      setLevelHintsUsed(prev => prev + 1);
   243→      const newRevealed = new Set(revealedLetters);
   244→      newRevealed.add(emptyIdx);
   245→      setRevealedLetters(newRevealed);
   246→      playSound("hint");
   247→      toast({ title: "💡 كشف حرف", description: `الحرف ${emptyIdx + 1}: ${currentWordText[emptyIdx]}` });
   248→    }
   249→  };
   250→
   251→  if (loading) {
   252→    return (
   253→      <div className="min-h-screen flex flex-col items-center justify-center gap-4 bg-gradient-to-b from-teal-900 to-cyan-950">
   254→        <motion.div
   255→          animate={{ rotate: 360 }}
   256→          transition={{ duration: 1, repeat: Infinity, ease: "linear" }}
   257→          className="w-16 h-16 rounded-full border-4 border-teal-400 border-t-transparent"
   258→        />
   259→        <p className="text-teal-200 text-sm">جاري تحميل اللغز...</p>
   260→      </div>
   261→    );
   262→  }
   263→
   264→  if (!level || !currentWord) {
   265→    return (
   266→      <div className="flex items-center justify-center min-h-screen bg-gradient-to-b from-teal-900 to-cyan-950">
   267→        <Card className="p-8 text-center bg-white/10 border-white/20 backdrop-blur-md">
   268→          <p className="text-white mb-4">المرحلة غير موجودة</p>
   269→          <Button onClick={() => onExit?.() || router.push("/")} variant="outline">العودة</Button>
   270→        </Card>
   271→      </div>
   272→    );
   273→  }
   274→
   275→  const puzzleType = level.puzzleType || "text";
   276→  const progress = (foundWords.length / level.words.length) * 100;
   277→  const difficultyLabels: Record<string, { label: string; color: string }> = {
   278→    easy: { label: "سهل", color: "bg-green-500/20 text-green-300" },
   279→    medium: { label: "متوسط", color: "bg-amber-500/20 text-amber-300" },
   280→    hard: { label: "صعب", color: "bg-red-500/20 text-red-300" },
   281→  };
   282→
   283→  return (
   284→    <div className="min-h-screen bg-gradient-to-b from-teal-800 via-teal-900 to-cyan-950 flex flex-col pattern-pro">
   285→      {/* شريط علوي محسّن */}
   286→      <header className="bg-teal-950/60 backdrop-blur-xl border-b border-white/10 sticky top-0 z-30">
   287→        <div className="max-w-4xl mx-auto px-4 py-3 flex items-center justify-between">
   288→          <button
   289→            onClick={() => { playSound("buttonClick"); if (onExit) onExit(); else router.push("/"); }}
   290→            className="flex items-center gap-2 text-white/80 hover:text-white px-3 py-1.5 rounded-lg hover:bg-white/10 transition-colors"
   291→          >
   292→            <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5">
   293→              <polyline points="15 18 9 12 15 6" />
   294→            </svg>
   295→            <span className="text-sm font-medium hidden sm:inline">رجوع</span>
   296→          </button>
   297→
   298→          <div className="flex items-center gap-3">
   299→            <span className="text-white font-bold text-lg">مرحلة {level.number}</span>
   300→            {difficultyLabels[level.difficulty] && (
   301→              <Badge className={cn("text-xs border-0", difficultyLabels[level.difficulty].color)}>
   302→                {difficultyLabels[level.difficulty].label}
   303→              </Badge>
   304→            )}
   305→          </div>
   306→
   307→          <div className="flex items-center gap-2">
   308→            <button onClick={toggleMute} className="p-2 rounded-lg hover:bg-white/10 transition-colors text-white/80">
   309→              {isMuted ? (
   310→                <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
   311→                  <polygon points="11 5 6 9 2 9 2 15 6 15 11 19 11 5" />
   312→                  <line x1="23" y1="9" x2="17" y2="15" />
   313→                  <line x1="17" y1="9" x2="23" y2="15" />
   314→                </svg>
   315→              ) : (
   316→                <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
   317→                  <polygon points="11 5 6 9 2 9 2 15 6 15 11 19 11 5" />
   318→                  <path d="M19.07 4.93a10 10 0 0 1 0 14.14M15.54 8.46a5 5 0 0 1 0 7.07" />
   319→                </svg>
   320→              )}
   321→            </button>
   322→            <div className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-white/10 backdrop-blur-sm">
   323→              <span className="text-yellow-400 text-sm">🪙</span>
   324→              <span className="text-sm font-bold text-white">{coins}</span>
   325→            </div>
   326→          </div>
   327→        </div>
   328→        {/* شريط التقدم */}
   329→        <div className="h-1 bg-teal-950/50">
   330→          <motion.div
   331→            className="h-full bg-gradient-to-r from-yellow-400 to-amber-500"
   332→            animate={{ width: `${progress}%` }}
   333→            transition={{ duration: 0.5, type: "spring", stiffness: 100 }}
   334→          />
   335→        </div>
   336→      </header>
   337→
   338→      {/* المحتوى الرئيسي */}
   339→      <main className="flex-1 max-w-4xl mx-auto w-full px-4 py-4 flex flex-col gap-4">
   340→        {/* مؤشر الأسئلة */}
   341→        <div className="flex items-center gap-2 justify-center">
   342→          {level.words.map((w, idx) => (
   343→            <div
   344→              key={idx}
   345→              className={cn(
   346→                "h-2 rounded-full transition-all",
   347→                foundWords.includes(w.text)
   348→                  ? "w-8 bg-green-400"
   349→                  : idx === currentWordIndex
   350→                  ? "w-8 bg-yellow-400 animate-pulse"
   351→                  : "w-2 bg-white/20"
   352→              )}
   353→            />
   354→          ))}
   355→        </div>
   356→
   357→        {/* منطقة اللغز */}
   358→        <motion.div
   359→          key={currentWordIndex}
   360→          initial={{ opacity: 0, y: 20 }}
   361→          animate={{ opacity: 1, y: 0 }}
   362→          className={cn("flex-1 flex flex-col items-center justify-center gap-4", wrongShake && "animate-shake")}
   363→          style={wrongShake ? { animation: "shake 0.5s" } : {}}
   364→        >
   365→          {/* 1. عرض الصورة إذا كان اللغز بصري */}
   366→          {puzzleType === "image" && currentWord.imageUrl && (
   367→            <div className="relative w-48 h-48 sm:w-56 sm:h-56 rounded-2xl overflow-hidden shadow-2xl border-2 border-white/20">
   368→              <img src={currentWord.imageUrl} alt="لغز" className="w-full h-full object-cover"
   369→                onError={(e) => { (e.target as HTMLImageElement).parentElement!.innerHTML = '<div class="w-full h-full flex items-center justify-center bg-teal-700 text-white text-6xl">🖼️</div>'; }}
   370→              />
   371→            </div>
   372→          )}
   373→
   374→          {/* 2. عرض 4 صور */}
   375→          {puzzleType === "images4" && (
   376→            <div className="grid grid-cols-2 gap-2 w-64 sm:w-72">
   377→              {[currentWord.imageUrl, currentWord.image2Url, currentWord.image3Url, currentWord.image4Url].map((url, i) => (
   378→                url ? (
   379→                  <div key={i} className="aspect-square rounded-xl overflow-hidden shadow-lg border border-white/10">
   380→                    <img src={url} alt={`صورة ${i+1}`} className="w-full h-full object-cover"
   381→                      onError={(e) => { (e.target as HTMLImageElement).parentElement!.innerHTML = '<div class="w-full h-full flex items-center justify-center bg-teal-700 text-white text-3xl">🖼️</div>'; }}
   382→                    />
   383→                  </div>
   384→                ) : (
   385→                  <div key={i} className="aspect-square rounded-xl bg-teal-700/50 flex items-center justify-center text-white/30 text-3xl">🖼️</div>
   386→                )
   387→              ))}
   388→            </div>
   389→          )}
   390→
   391→          {/* 3. عرض الإيموجي */}
   392→          {puzzleType === "emoji" && currentWord.emojis && (
   393→            <div className="bg-gradient-to-br from-amber-500/20 to-orange-500/20 rounded-2xl px-8 py-6 border-2 border-amber-400/30">
   394→              <div className="flex items-center justify-center gap-2 text-5xl sm:text-6xl">
   395→                {currentWord.emojis.split("").map((emoji, i) => (
   396→                  <motion.span
   397→                    key={i}
   398→                    initial={{ scale: 0, rotate: -180 }}
   399→                    animate={{ scale: 1, rotate: 0 }}
   400→                    transition={{ delay: i * 0.1, type: "spring" }}
   401→                  >
   402→                    {emoji}
   403→                  </motion.span>
   404→                ))}
   405→              </div>
   406→              {currentWord.clue && (
   407→                <p className="text-white/70 text-sm text-center mt-3">{currentWord.clue}</p>
   408→              )}
   409→            </div>
   410→          )}
   411→
   412→          {/* 4. عرض الحروف المبعثرة */}
   413→          {puzzleType === "scrambled" && currentWord.sequence && (
   414→            <div className="bg-gradient-to-br from-purple-500/20 to-blue-500/20 rounded-2xl px-6 py-4 border-2 border-purple-400/30 w-full max-w-md">
   415→              <p className="text-white/70 text-sm text-center mb-3">{currentWord.clue}</p>
   416→              <div className="flex items-center justify-center gap-2">
   417→                {currentWord.sequence.split("").map((letter, i) => (
   418→                  <motion.div
   419→                    key={i}
   420→                    initial={{ scale: 0, y: -20 }}
   421→                    animate={{ scale: 1, y: 0 }}
   422→                    transition={{ delay: i * 0.1, type: "spring" }}
   423→                    className="w-12 h-12 rounded-xl bg-white/20 border-2 border-purple-400/50 flex items-center justify-center text-white text-xl font-bold"
   424→                  >
   425→                    {letter}
   426→                  </motion.div>
   427→                ))}
   428→              </div>
   429→              <p className="text-white/50 text-xs text-center mt-3">↑ رتب هذه الحروف لتكوين كلمة</p>
   430→            </div>
   431→          )}
   432→
   433→          {/* 5. عرض الفراغات */}
   434→          {puzzleType === "fillblank" && currentWord.fillBlanks && (
   435→            <div className="bg-gradient-to-br from-teal-600 to-cyan-700 rounded-2xl px-6 py-4 shadow-xl max-w-md w-full">
   436→              <p className="text-white text-lg sm:text-xl font-bold text-center leading-relaxed" style={{ fontFamily: "var(--font-tajawal)" }}>
   437→                {currentWord.fillBlanks.split("___").map((part, i, arr) => (
   438→                  <span key={i}>
   439→                    {part}
   440→                    {i < arr.length - 1 && (
   441→                      <span className="inline-block mx-1 px-3 py-0.5 rounded-lg bg-yellow-400/30 border-b-2 border-yellow-400 text-yellow-200">
   442→                        {"?".repeat(currentWord.text.length)}
   443→                      </span>
   444→                    )}
   445→                  </span>
   446→                ))}
   447→              </p>
