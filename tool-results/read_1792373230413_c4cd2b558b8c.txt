     1→"use client";
     2→
     3→import { useState, useEffect } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import { Button } from "@/components/ui/button";
     6→import { Card } from "@/components/ui/card";
     7→import { Input } from "@/components/ui/input";
     8→import { Label } from "@/components/ui/label";
     9→import { Badge } from "@/components/ui/badge";
    10→import { GridEditor } from "./grid-editor";
    11→import { QuizEditor } from "./quiz-editor";
    12→import { useToast } from "@/hooks/use-toast";
    13→import { cn } from "@/lib/utils";
    14→import { PUZZLE_TYPES, PuzzleType, getPuzzleTypeDef } from "@/lib/puzzle-types";
    15→
    16→interface LevelEditorProps {
    17→  levelId: number | null;
    18→  chapterId: number;
    19→  onBack: () => void;
    20→}
    21→
    22→interface Cell {
    23→  row: number;
    24→  col: number;
    25→  letter: string;
    26→  number?: number | null;
    27→}
    28→
    29→interface Word {
    30→  id?: number;
    31→  text: string;
    32→  clue: string;
    33→  category?: string;
    34→  imageUrl?: string;
    35→  image2Url?: string;
    36→  image3Url?: string;
    37→  image4Url?: string;
    38→  direction: "across" | "down";
    39→  startRow: number;
    40→  startCol: number;
    41→  number: number;
    42→}
    43→
    44→const CATEGORIES = [
    45→  { value: "", label: "بدون تصنيف", icon: "📝" },
    46→  { value: "من", label: "من؟", icon: "👤" },
    47→  { value: "ماذا", label: "ماذا؟", icon: "❓" },
    48→  { value: "أين", label: "أين؟", icon: "📍" },
    49→  { value: "متى", label: "متى؟", icon: "📅" },
    50→  { value: "كيف", label: "كيف؟", icon: "💡" },
    51→  { value: "كم", label: "كم؟", icon: "🔢" },
    52→  { value: "أكمل", label: "أكمل الجملة", icon: "✍️" },
    53→  { value: "عرف", label: "عرّف", icon: "📖" },
    54→  { value: "ضد", label: "ضد (مفردات)", icon: "⚖️" },
    55→  { value: "مرادف", label: "مرادف", icon: "🔗" },
    56→];
    57→
    58→export function LevelEditor({ levelId, chapterId, onBack }: LevelEditorProps) {
    59→  const [gameMode, setGameMode] = useState<"crossword" | "quiz">("quiz");
    60→  const [puzzleType, setPuzzleType] = useState<PuzzleType>("text");
    61→  const [title, setTitle] = useState("");
    62→  const [difficulty, setDifficulty] = useState<"easy" | "medium" | "hard">("easy");
    63→  const [gridSizeRows, setGridSizeRows] = useState(5);
    64→  const [gridSizeCols, setGridSizeCols] = useState(5);
    65→  const [availableLetters, setAvailableLetters] = useState<string[]>([]);
    66→  const [bonusWords, setBonusWords] = useState<string[]>([]);
    67→  const [reward, setReward] = useState(50);
    68→  const [hintCost, setHintCost] = useState(10);
    69→  const [cells, setCells] = useState<Cell[]>([]);
    70→  const [words, setWords] = useState<Word[]>([]);
    71→  const [loading, setLoading] = useState(false);
    72→  const [saving, setSaving] = useState(false);
    73→  const [validation, setValidation] = useState<{ valid: boolean; errors: string[] } | null>(null);
    74→  const [activeTab, setActiveTab] = useState<"editor" | "settings" | "preview">("editor");
    75→  const { toast } = useToast();
    76→
    77→  useEffect(() => {
    78→    if (!levelId) {
    79→      setGameMode("quiz");
    80→      return;
    81→    }
    82→    setLoading(true);
    83→    fetch(`/api/admin/levels/${levelId}`)
    84→      .then((res) => res.json())
    85→      .then((data) => {
    86→        const level = data.level;
    87→        if (level) {
    88→          setGameMode(level.gameMode || "quiz");
    89→          setTitle(level.title);
    90→          setDifficulty(level.difficulty);
    91→          setGridSizeRows(level.gridSizeRows);
    92→          setGridSizeCols(level.gridSizeCols);
    93→          setAvailableLetters(level.availableLetters || []);
    94→          setBonusWords(level.bonusWords || []);
    95→          setReward(level.reward);
    96→          setHintCost(level.hintCost);
    97→          setCells(level.cells || []);
    98→          setWords(level.words || []);
    99→        }
   100→      })
   101→      .catch((err) => {
   102→        console.error("Error loading level:", err);
   103→        toast({ title: "خطأ", description: "فشل تحميل المرحلة", variant: "destructive" });
   104→      })
   105→      .finally(() => setLoading(false));
   106→  }, [levelId, toast]);
   107→
   108→  // توليد الحروف المتاحة تلقائياً من الكلمات
   109→  const generateLetters = (allWords: Word[], extraWords: string[] = []): string[] => {
   110→    const letterCount = new Map<string, number>();
   111→    for (const word of allWords) {
   112→      for (const letter of word.text) {
   113→        letterCount.set(letter, (letterCount.get(letter) || 0) + 1);
   114→      }
   115→    }
   116→    const letters: string[] = [];
   117→    letterCount.forEach((count, letter) => {
   118→      for (let i = 0; i < count; i++) letters.push(letter);
   119→    });
   120→    const arabicLetters = "ابتثجدحخدرزسشصضطظعغفقكلمنهوي";
   121→    const extraCount = Math.max(3, Math.ceil(letters.length * 0.3));
   122→    for (let i = 0; i < extraCount; i++) {
   123→      const randomLetter = arabicLetters[Math.floor(Math.random() * arabicLetters.length)];
   124→      if (!letters.includes(randomLetter)) letters.push(randomLetter);
   125→    }
   126→    return letters.sort(() => Math.random() - 0.5);
   127→  };
   128→
   129→  const handleWordsChange = (newWords: Word[], newCells?: Cell[]) => {
   130→    setWords(newWords);
   131→    if (newCells) setCells(newCells);
   132→    setAvailableLetters(generateLetters(newWords, bonusWords));
   133→  };
   134→
   135→  const handleGridChange = (newCells: Cell[], newWords: Word[]) => {
   136→    setCells(newCells);
   137→    setWords(newWords);
   138→    setAvailableLetters(generateLetters(newWords, bonusWords));
   139→  };
   140→
   141→  const handleAddBonusWord = (word: string) => {
   142→    if (word.trim() && !bonusWords.includes(word.trim())) {
   143→      const newBonus = [...bonusWords, word.trim()];
   144→      setBonusWords(newBonus);
   145→      setAvailableLetters(generateLetters(words, newBonus));
   146→    }
   147→  };
   148→
   149→  const handleRemoveBonusWord = (word: string) => {
   150→    const newBonus = bonusWords.filter((w) => w !== word);
   151→    setBonusWords(newBonus);
   152→    setAvailableLetters(generateLetters(words, newBonus));
   153→  };
   154→
   155→  const handleSave = async (publish: boolean = false) => {
   156→    setSaving(true);
   157→    setValidation(null);
   158→
   159→    if (words.length === 0) {
   160→      toast({ title: "لا يمكن الحفظ", description: "أضف كلمة واحدة على الأقل", variant: "destructive" });
   161→      setSaving(false);
   162→      return;
   163→    }
   164→
   165→    // توليد الحروف تلقائياً إذا كانت فارغة
   166→    let lettersToSend = availableLetters;
   167→    if (!lettersToSend || lettersToSend.length === 0) {
   168→      const answerWords = words.map((w) => w.text);
   169→      lettersToSend = generateLetters(words, bonusWords);
   170→      setAvailableLetters(lettersToSend);
   171→    }
   172→
   173→    // استخدام نوع اللغز المختار من المحرر
   174→    const payload = {
   175→      chapterId,
   176→      gameMode,
   177→      puzzleType: puzzleType,
   178→      title,
   179→      difficulty,
   180→      gridSizeRows,
   181→      gridSizeCols,
   182→      availableLetters: lettersToSend,
   183→      bonusWords,
   184→      reward,
   185→      hintCost,
   186→      words: words.map((w) => ({
   187→        text: w.text,
   188→        clue: w.clue,
   189→        category: w.category || "",
   190→        imageUrl: w.imageUrl || "",
   191→        image2Url: w.image2Url || "",
   192→        image3Url: w.image3Url || "",
   193→        image4Url: w.image4Url || "",
   194→        direction: w.direction,
   195→        startRow: w.startRow,
   196→        startCol: w.startCol,
   197→        number: w.number,
   198→      })),
   199→      cells: gameMode === "crossword" ? cells : [],
   200→    };
   201→
   202→    try {
   203→      const url = levelId ? `/api/admin/levels/${levelId}` : "/api/admin/levels";
   204→      const method = levelId ? "PUT" : "POST";
   205→      const response = await fetch(url, {
   206→        method,
   207→        headers: { "Content-Type": "application/json" },
   208→        body: JSON.stringify(payload),
   209→      });
   210→      const data = await response.json();
   211→
   212→      if (!response.ok) {
   213→        if (data.details) {
   214→          setValidation({ valid: false, errors: data.details });
   215→          toast({ title: "فشل التحقق", description: data.details.join(" · "), variant: "destructive" });
   216→        } else {
   217→          toast({ title: "فشل الحفظ", description: data.error || "حدث خطأ غير متوقع", variant: "destructive" });
   218→        }
   219→        setSaving(false);
   220→        return;
   221→      }
   222→
   223→      toast({
   224→        title: levelId ? "تم التحديث" : "تم الإنشاء",
   225→        description: publish ? "تم نشر المرحلة" : "تم حفظ المرحلة كمسودة",
   226→      });
   227→
   228→      if (publish && data.level?.id) {
   229→        const publishResponse = await fetch(`/api/admin/levels/${data.level.id}/publish`, { method: "POST" });
   230→        if (publishResponse.ok) toast({ title: "تم النشر", description: "المرحلة منشورة الآن" });
   231→      }
   232→      onBack();
   233→    } catch (error) {
   234→      console.error("Error saving level:", error);
   235→      toast({ title: "خطأ", description: "فشل حفظ المرحلة", variant: "destructive" });
   236→    } finally {
   237→      setSaving(false);
   238→    }
   239→  };
   240→
   241→  if (loading) {
   242→    return (
   243→      <div className="min-h-screen flex items-center justify-center bg-stone-50">
   244→        <div className="w-12 h-12 border-4 border-amber-200 border-t-amber-500 rounded-full animate-spin" />
   245→      </div>
   246→    );
   247→  }
   248→
   249→  const canSave = title && words.length > 0;
   250→
   251→  // تحديد نوع اللغز للعرض
   252→  const puzzleTypeDef = getPuzzleTypeDef(puzzleType);
   253→
   254→  return (
   255→    <div className="min-h-screen bg-stone-50">
   256→      {/* ===== الرأس ===== */}
   257→      <header className="bg-white border-b border-stone-200 sticky top-0 z-30 shadow-sm">
   258→        <div className="max-w-6xl mx-auto px-4 py-3">
   259→          <div className="flex items-center justify-between gap-3">
   260→            {/* يمين: زر الرجوع + العنوان */}
   261→            <div className="flex items-center gap-3 min-w-0">
   262→              <Button variant="ghost" size="sm" onClick={onBack} className="shrink-0">
   263→                <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5">
   264→                  <polyline points="15 18 9 12 15 6" />
   265→                </svg>
   266→              </Button>
   267→              <div className="min-w-0">
   268→                <h1 className="text-lg font-bold text-stone-800 truncate">
   269→                  {levelId ? "تعديل المرحلة" : "مرحلة جديدة"}
   270→                </h1>
   271→                <div className="flex items-center gap-2 text-xs text-stone-500">
   272→                  <Badge variant="outline" className="text-[10px] py-0">
   273→                    {gameMode === "quiz" ? "❓ أسئلة" : "🔤 شبكة"}
   274→                  </Badge>
   275→                  {puzzleType !== "text" && (
   276→                    <Badge variant="outline" className="text-[10px] py-0">
   277→                      {puzzleTypeDef?.icon} {puzzleTypeDef?.label}
   278→                    </Badge>
   279→                  )}
   280→                  <span>{words.length} سؤال</span>
   281→                </div>
   282→              </div>
   283→            </div>
   284→
   285→            {/* يسار: أزرار الحفظ */}
   286→            <div className="flex items-center gap-2 shrink-0">
   287→              {/* مؤشر الحالة */}
   288→              {!canSave && (
   289→                <span className="text-xs text-amber-600 font-medium hidden md:flex items-center gap-1">
   290→                  <span className="w-2 h-2 rounded-full bg-amber-400 animate-pulse" />
   291→                  {!title ? "أدخل العنوان" : "أضف سؤالاً"}
   292→                </span>
   293→              )}
   294→              {canSave && (
   295→                <span className="text-xs text-green-600 font-medium hidden md:flex items-center gap-1">
   296→                  <span className="w-2 h-2 rounded-full bg-green-500" />
   297→                  جاهز للحفظ
   298→                </span>
   299→              )}
   300→
   301→              <Button
   302→                variant="outline"
   303→                size="sm"
   304→                onClick={() => handleSave(false)}
   305→                disabled={saving || !canSave}
   306→                className={cn(!canSave && "opacity-40")}
   307→              >
   308→                {saving ? "⏳" : "💾"} مسودة
   309→              </Button>
   310→              <Button
   311→                size="sm"
   312→                onClick={() => handleSave(true)}
   313→                disabled={saving || !canSave}
   314→                className={cn(
   315→                  "bg-gradient-to-r from-green-500 to-green-600 hover:from-green-600 hover:to-green-700",
   316→                  !canSave && "opacity-40"
   317→                )}
   318→              >
   319→                {saving ? "⏳ جاري..." : "🚀 حفظ ونشر"}
   320→              </Button>
   321→            </div>
   322→          </div>
   323→
   324→          {/* تبويبات */}
   325→          <div className="flex items-center gap-1 mt-3 border-b border-stone-200 -mb-3">
   326→            {([
   327→              { id: "editor", label: "المحرر", icon: "✏️" },
   328→              { id: "settings", label: "الإعدادات", icon: "⚙️" },
   329→              { id: "preview", label: "معاينة", icon: "👁️" },
   330→            ] as const).map((tab) => (
   331→              <button
   332→                key={tab.id}
   333→                onClick={() => setActiveTab(tab.id)}
   334→                className={cn(
   335→                  "px-4 py-2 text-sm font-medium border-b-2 transition-all flex items-center gap-1.5 -mb-px",
   336→                  activeTab === tab.id
   337→                    ? "border-amber-500 text-amber-700"
   338→                    : "border-transparent text-stone-500 hover:text-stone-700"
   339→                )}
   340→              >
   341→                <span>{tab.icon}</span>
   342→                <span>{tab.label}</span>
   343→              </button>
   344→            ))}
   345→          </div>
   346→        </div>
   347→      </header>
   348→
   349→      {/* ===== المحتوى ===== */}
   350→      <div className="max-w-6xl mx-auto px-4 py-6">
   351→        {/* تبويب: المحرر */}
   352→        {activeTab === "editor" && (
   353→          <div className="space-y-4">
   354→            {/* اختيار نوع اللغز */}
   355→            <Card className="p-4">
   356→              <Label className="text-sm font-bold text-stone-700 mb-3 block">
   357→                🎮 نوع اللغز
   358→              </Label>
   359→              <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-2">
   360→                {PUZZLE_TYPES.map((pt) => (
   361→                  <button
   362→                    key={pt.type}
   363→                    onClick={() => {
   364→                      setPuzzleType(pt.type);
   365→                      if (pt.type === "crossword") setGameMode("crossword");
   366→                      else setGameMode("quiz");
   367→                    }}
   368→                    className={cn(
   369→                      "py-3 px-2 rounded-xl text-sm font-bold border-2 transition-all flex flex-col items-center gap-1 text-center",
   370→                      puzzleType === pt.type
   371→                        ? "bg-teal-50 border-teal-400 text-teal-800 shadow-sm"
   372→                        : "bg-white border-stone-200 text-stone-600 hover:border-stone-300"
   373→                    )}
   374→                    style={puzzleType === pt.type ? { borderColor: pt.color } : {}}
   375→                  >
   376→                    <span className="text-2xl">{pt.icon}</span>
   377→                    <span className="text-xs">{pt.label}</span>
   378→                    <span className="text-[9px] text-stone-400">{pt.difficulty}</span>
   379→                  </button>
   380→                ))}
   381→              </div>
   382→              {/* وصف النوع المختار */}
   383→              {puzzleType !== "text" && (
   384→                <div className="mt-3 p-2 rounded-lg bg-stone-50 border border-stone-200 flex items-center gap-2">
   385→                  <span className="text-lg">{getPuzzleTypeDef(puzzleType)?.icon}</span>
   386→                  <div>
   387→                    <p className="text-xs font-bold text-stone-700">{getPuzzleTypeDef(puzzleType)?.label}</p>
   388→                    <p className="text-xs text-stone-500">{getPuzzleTypeDef(puzzleType)?.description}</p>
   389→                  </div>
   390→                </div>
   391→              )}
   392→            </Card>
   393→
   394→            {/* اختيار الصعوبة */}
   395→            <Card className="p-4">
   396→              <Label className="text-sm font-bold text-stone-700 mb-3 block">
   397→                📊 مستوى الصعوبة
   398→              </Label>
   399→              <div className="grid grid-cols-3 gap-2">
   400→                {([
   401→                  { val: "easy", label: "سهل", icon: "🟢", reward: 50 },
   402→                  { val: "medium", label: "متوسط", icon: "🟡", reward: 75 },
   403→                  { val: "hard", label: "صعب", icon: "🔴", reward: 100 },
   404→                ] as const).map((d) => (
   405→                  <button
   406→                    key={d.val}
   407→                    onClick={() => {
   408→                      setDifficulty(d.val);
   409→                      setReward(d.reward);
   410→                    }}
   411→                    className={cn(
   412→                      "py-2 rounded-lg text-sm font-medium border-2 transition-all flex flex-col items-center gap-0.5",
   413→                      difficulty === d.val
   414→                        ? "bg-amber-50 border-amber-400 text-amber-800"
   415→                        : "bg-white border-stone-200 text-stone-600"
   416→                    )}
   417→                  >
   418→                    <span>{d.icon}</span>
   419→                    <span>{d.label}</span>
   420→                    <span className="text-[10px] text-stone-500">{d.reward} 🪙</span>
   421→                  </button>
   422→                ))}
   423→              </div>
   424→            </Card>
   425→
   426→            {/* المحرر حسب الوضع */}
   427→            {gameMode === "quiz" ? (
   428→              <QuizEditor
   429→                words={words}
   430→                onChange={handleWordsChange}
   431→                categories={CATEGORIES}
   432→              />
   433→            ) : (
   434→              <Card className="p-4">
   435→                <div className="flex items-center justify-between mb-3">
   436→                  <h3 className="font-bold text-stone-700">محرر الشبكة المتقاطعة</h3>
   437→                  <span className="text-xs text-stone-500">اسحب من خلية إلى أخرى لرسم كلمة</span>
   438→                </div>
   439→                <GridEditor
   440→                  rows={gridSizeRows}
   441→                  cols={gridSizeCols}
   442→                  cells={cells}
   443→                  words={words}
   444→                  onChange={handleGridChange}
   445→                />
   446→              </Card>
   447→            )}
   448→
   449→            {/* نتائج الفحص */}
   450→            {validation && !validation.valid && (
   451→              <Card className="p-4 border-red-200 bg-red-50">
   452→                <h3 className="font-bold text-red-700 mb-2">⚠️ أخطاء في الفحص</h3>
   453→                <ul className="space-y-1 text-sm text-red-600">
   454→                  {validation.errors.map((error, idx) => (
   455→                    <li key={idx}>• {error}</li>
   456→                  ))}
   457→                </ul>
   458→              </Card>
   459→            )}
   460→          </div>
   461→        )}
   462→
   463→        {/* تبويب: الإعدادات */}
   464→        {activeTab === "settings" && (
   465→          <div className="space-y-4 max-w-2xl mx-auto">
   466→            {/* معلومات أساسية */}
   467→            <Card className="p-5">
   468→              <h3 className="font-bold text-stone-700 mb-4 flex items-center gap-2">
   469→                📝 معلومات المرحلة
   470→              </h3>
   471→              <div className="space-y-4">
   472→                <div>
   473→                  <Label className="text-sm font-medium text-stone-600 mb-1.5 block">
   474→                    عنوان المرحلة
   475→                  </Label>
   476→                  <Input
   477→                    value={title}
   478→                    onChange={(e) => setTitle(e.target.value)}
   479→                    placeholder="مثال: لغز رقم 1"
   480→                    className="text-base"
   481→                  />
   482→                </div>
   483→
   484→                <div className="grid grid-cols-2 gap-3">
   485→                  <div>
   486→                    <Label className="text-sm font-medium text-stone-600 mb-1.5 block">
   487→                      المكافأة (🪙)
   488→                    </Label>
   489→                    <Input
   490→                      type="number"
   491→                      value={reward}
   492→                      onChange={(e) => setReward(parseInt(e.target.value) || 50)}
   493→                      min={0}
   494→                    />
   495→                  </div>
   496→                  <div>
   497→                    <Label className="text-sm font-medium text-stone-600 mb-1.5 block">
   498→                      تكلفة التلميح (🪙)
   499→                    </Label>
   500→                    <Input
   501→                      type="number"
   502→                      value={hintCost}
   503→                      onChange={(e) => setHintCost(parseInt(e.target.value) || 10)}
   504→                      min={0}
   505→                    />
   506→                  </div>
   507→                </div>
   508→
